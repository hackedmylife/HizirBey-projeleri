cmake_minimum_required(VERSION 3.22)
project(hizir_speech LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor" whisper-build)
add_library(hizir_speech SHARED speech.cpp)
target_compile_features(hizir_speech PRIVATE cxx_std_17)
target_link_libraries(hizir_speech PRIVATE whisper android log)
target_link_options(hizir_speech PRIVATE "-Wl,-z,max-page-size=16384")
